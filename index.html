Aggiungi colonna public_code (ID ATLETA, ID SCOUT, ID CLUB) alter table public.profiles add column if not exists public_code text unique; alter table public.players add column if not exists public_code text unique; -- 2. Genera automaticamente gli ID per i record già esistenti update public.profiles set public_code = ( case role when 'scout' then 'SCT-' when 'club' then 'CLB-' when 'admin' then 'ADM-' else 'USR-' end || upper(substr(md5(id::text), 1, 5)) ) where public_code is null; update public.players set public_code = 'ATL-' || upper(substr(md5(id), 1, 5)) where public_code is null; -- 3. Funzione sicura per permettere a un Calciatore/Genitore di collegare una scheda creata da uno Scout create or replace function public.claim_player_by_code(p_code text) returns text as $$ declare v_player_id text; v_owner uuid; begin select id, owner_user_id into v_player_id, v_owner from public.players where upper(public_code) = upper(trim(p_code)); if v_player_id is null then raise exception 'ID Atleta non trovato.'; end if; if v_owner is not null and v_owner <> auth.uid() then raise exception 'Questa scheda è già collegata a un altro account.'; end if; update public.players set owner_user_id = auth.uid(), privacy_consent = true, consent_type = 'parent' where id = v_player_id; return v_player_id; end; $$ language plpgsql security definer; grant execute on function public.claim_player_by_code(text) to authenticated;
